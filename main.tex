\begin{frame}{Dataset Quality Assesment}
\small
    \begin{itemize}
        \item Sampling Balance Indicator (SBI)
            \begin{align*}
                \mathrm{SBI} = \frac{1}{N^{(k)}}\sum_{i=1}^{N}\sum_{j\in\mathcal{N}_i^{(k)}} R_{ij}.
            \end{align*}
        where
        \begin{align*}
            R_{ij} = \frac{\omega_{ij}}{VD_{ij}}, \quad j \in \mathcal{N}_i^{(k)},
        \end{align*}

        \item Undersampling aware score (S)

        \begin{equation*}
            S_i = \max_{j} \left( 
            \frac{\omega_{ij}}{\omega_{\max}} 
            \cdot 
            \frac{VD_{ij}}{VD_{\max}}
            \right),
        \end{equation*}
    \end{itemize}
